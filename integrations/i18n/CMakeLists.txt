cmake_minimum_required(VERSION 3.20)
project(SpectrumI18n LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
add_library(spectrumi18n MODULE translator.cpp)
target_link_libraries(spectrumi18n PRIVATE Qt6::Core Qt6::Qml)
set_target_properties(spectrumi18n PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/Spectrum/I18n")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/qml/Spectrum/I18n")
configure_file(qmldir "${CMAKE_BINARY_DIR}/qml/Spectrum/I18n/qmldir" COPYONLY)
install(TARGETS spectrumi18n LIBRARY DESTINATION lib/qt6/qml/Spectrum/I18n)
install(FILES qmldir DESTINATION lib/qt6/qml/Spectrum/I18n)
